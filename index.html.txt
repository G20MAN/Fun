    <script>
        // 初始化画布
        const canvas = document.getElementById('canvas');
        const ctx = canvas.getContext('2d');
        let particles = [];
        const blessings = ["🥮", "🌕", "🎑", "✨", "中秋快乐", "团圆", "幸福", "阖家欢乐"];

        function resize() {
            canvas.width = window.innerWidth;
            canvas.height = window.innerHeight;
        }
        window.addEventListener('resize', resize);
        resize();

        // 动态生成抖音风格孔明灯背景
        for(let i=0; i<25; i++) {
            let lantern = document.createElement('div');
            lantern.className = 'lantern';
            lantern.style.left = Math.random() * 100 + 'vw';
            lantern.style.animationDuration = (Math.random() * 8 + 8) + 's';
            lantern.style.animationDelay = (Math.random() * 6) + 's';
            lantern.style.setProperty('--w', (Math.random() * 100 - 50) + 'px');
            document.body.appendChild(lantern);
        }

        // 粒子类定义 (文字+彩色碎星)
        class Particle {
            constructor(x, y) {
                this.x = x;
                this.y = y;
                this.isText = Math.random() > 0.4;
                this.text = blessings[Math.floor(Math.random() * blessings.length)];
                this.radius = Math.random() * 4 + 2;
                
                const angle = Math.random() * Math.PI * 2;
                const speed = Math.random() * 6 + 4;
                this.vx = Math.cos(angle) * speed;
                this.vy = Math.sin(angle) * speed;
                
                this.gravity = 0.12;
                this.opacity = 1;
                this.color = `hsla(${Math.random() * 45 + 35}, 100%, 60%, 1)`; // 金黄色系为主
            }

            draw() {
                ctx.save();
                ctx.globalAlpha = this.opacity;
                if(this.isText) {
                    ctx.font = 'bold 16px sans-serif';
                    ctx.fillStyle = this.color;
                    ctx.fillText(this.text, this.x, this.y);
                } else {
                    ctx.beginPath();
                    ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
                    ctx.fillStyle = '#ffd700';
                    ctx.shadowBlur = 10;
                    ctx.shadowColor = '#ff8c00';
                    ctx.fill();
                }
                ctx.restore();
            }

            update() {
                this.vx *= 0.98;
                this.vy *= 0.98;
                this.vy += this.gravity;
                this.x += this.vx;
                this.y += this.vy;
                this.opacity -= 0.015;
            }
        }

        // 点击月饼触发爆炸动效
        function explode(e) {
            const rect = e.currentTarget.getBoundingClientRect();
            const x = rect.left + rect.width / 2;
            const y = rect.top + rect.height / 2;
            
            // 一次性产生大量粒子
            for (let i = 0; i < 40; i++) {
                particles.push(new Particle(x, y));
            }
        }

        // 动画渲染主循环
        function loop() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            
            // 绘制星空底色微光
            for (let i = particles.length - 1; i >= 0; i--) {
                particles[i].update();
                particles[i].draw();
                if (particles[i].opacity <= 0) {
                    particles.splice(i, 1);
                }
            }
            requestAnimationFrame(loop);
        }
        loop();
    </script>
